process_chromhmm_raw <- function(chrom_gr, roi_gr) {

  hits <- findOverlaps(roi_gr, chrom_gr, ignore.strand = TRUE)

  peak_states <- data.frame(
    peak_id   = queryHits(hits),
    Peak_name = mcols(roi_gr)$name[queryHits(hits)],
    state     = mcols(chrom_gr)$name[subjectHits(hits)],
    width     = width(pintersect(
      roi_gr[queryHits(hits)],
      chrom_gr[subjectHits(hits)]
    ))
  )

  peak_states_summed <- peak_states %>%
    group_by(Peak_name, state) %>%
    summarize(total_width = sum(width), .groups = "drop")

  peak_states_props <- peak_states_summed %>%
    mutate(frac = total_width / 401)
  
  return(peak_states_props)
}

process_chromhmm_KZFP_raw <- function(chrom_gr, roi_gr) {
seqlevelsStyle(roi_gr) <- seqlevelsStyle(chrom_gr)
  hits <- findOverlaps(roi_gr, chrom_gr, ignore.strand = TRUE)

  tss_states <- data.frame(
    tx_id_num   = queryHits(hits),
    Tx_ID = mcols(roi_gr)$name[queryHits(hits)],
    state     = mcols(chrom_gr)$name[subjectHits(hits)],
    width     = width(pintersect(
      roi_gr[queryHits(hits)],
      chrom_gr[subjectHits(hits)]
    ))
  )

  tss_states_summed <- tss_states %>%
    group_by(Tx_ID, state) %>%
    summarize(total_width = sum(width), .groups = "drop")

  # compute actual ROI widths per Tx_ID (handles variable windows correctly)
  roi_widths <- data.frame(
    Tx_ID = mcols(roi_gr)$name,
    roi_width = width(roi_gr)
  ) %>%
    dplyr::distinct()

  peak_states_props <- tss_states_summed %>%
    dplyr::left_join(roi_widths, by = "Tx_ID") %>%
    dplyr::mutate(frac = total_width / roi_width)

   return(peak_states_props)
}

process_chromhmm_TE_raw <- function(chrom_gr, te_gr, te_name) {

  hits <- findOverlaps(te_gr, chrom_gr, ignore.strand = TRUE)

  te_states <- data.frame(
    te_id    = mcols(te_gr)$name[queryHits(hits)],
    te_name  = te_name,
    state    = mcols(chrom_gr)$name[subjectHits(hits)],
    width    = width(pintersect(
      te_gr[queryHits(hits)],
      chrom_gr[subjectHits(hits)]
    )),
    TE_width = width(te_gr[queryHits(hits)])
  )

  te_states_summed <- te_states %>%
    group_by(te_id, te_name, state, TE_width) %>%
    summarize(total_width = sum(width), .groups = "drop")

  te_states_props <- te_states_summed %>%
    mutate(frac = total_width / TE_width)
  
  return(te_states_props)
}

plot_trajectory <- function(dominant_df, TE, id_col="te_id",state_col="Broad_state") {
  
  state_colors <- c(
  "Transcription" = "#D73027",
  "Enhancer" = "#FC8D59",
  "Poised" = "#FEE08B",
  "Repressed" = "#4575B4",
  "Quiescent" = "#D9D9D9")
  
  name_color_key <- dominant_df %>%
  dplyr::distinct(Name, Broad_state) %>%
  dplyr::filter(!is.na(Name), !is.na(Broad_state)) %>%
  dplyr::mutate(color = unname(state_colors[Broad_state]))

name_colors <- stats::setNames(
  name_color_key$color,
  name_color_key$Name)

if (state_col == "Broad_state") {
  plot_colors <- state_colors
} else if (state_col == "Name") {
  plot_colors <- name_colors
}

  trajectory <- dominant_df %>%
    dplyr::mutate(Condition=factor(Condition, levels=c("VEH","DOX"))) %>% 
  dplyr::select(all_of(id_col),te_name, Condition, Timepoint, all_of(state_col)) %>%
  tidyr::pivot_wider(
    names_from = Timepoint,
    values_from = all_of(state_col))
  
  
  ## checking if TE has all markings
  trajectory <- trajectory %>% 
    mutate(status=case_when(
        rowSums(is.na(across(c(`24T`, `24R`, `144R`)))) == 0 ~ "complete",
        rowSums(is.na(across(c(`24T`, `24R`, `144R`)))) == 3 ~ "missing",
        TRUE ~ "partial"))
  
  n_elements <- trajectory %>%
  distinct(.data[[id_col]]) %>%
  nrow()
  
  trajectory_complete <- trajectory %>%
    filter(status == "complete")
  
   alluvial_plot <- ggplot(trajectory_complete, aes(axis1 = `24T`,
                                                    axis2 = `24R`,
                                                    axis3 = `144R`,
                                                    y = 1)) +
     geom_alluvium(aes(fill = after_stat(stratum)),
                   alpha = 0.4) +
     geom_stratum(aes(fill = after_stat(stratum)),
                  color = "black") +
     geom_text(stat = "stratum",aes(label = after_stat(stratum)),
               size = 3) +
     facet_wrap(~ Condition) +
     scale_x_discrete(limits = c("24T", "24R", "144R"),
                      expand = c(0.1, 0.05)) +
     scale_fill_manual(values = plot_colors,
                       na.value = "grey80") +
     labs(title = paste0("Complete ChromHMM state trajectories for ", TE," (n = ",n_elements," integrants)"),
          x = NULL,
          y = paste0("Number of ", TE, " elements"),
          fill = "State") +
     theme_minimal()
   #### calculating incompletes and QC
    qc_counts <- trajectory %>%
    count(Condition, status)
    
    qc_plot <- ggplot(qc_counts, aes(x = status, y = n, fill = status)) +
    geom_col(width = 0.6) +
    facet_wrap(~ Condition) +
    theme_minimal() +
    labs( x = NULL,
          y = paste0("Number of ", TE, " elements"),
          title = paste0("ChromHMM trajectory completeness for ", TE)) +
      theme(legend.position = "none")
    
    qc_summary <- trajectory %>%
    group_by(Condition) %>%
    summarise(n_elements = dplyr::n(),
              complete = sum(status == "complete"),
              partial = sum(status == "partial"),
              missing = sum(status == "missing"),
              frac_complete = complete / n_elements,
              .groups = "drop")
    
    return(list(
    alluvial_plot = alluvial_plot,
    qc_plot = qc_plot,
    trajectory = trajectory,
    trajectory_complete = trajectory_complete,
    QC_counts = qc_counts,
    QC_summary = qc_summary
  ))
}
