KZFP_gr <- GRanges(seqnames = human_protein_coding_KZFPs$chrom,
                   ranges= IRanges(start=human_protein_coding_KZFPs$start+1,
                                   end= human_protein_coding_KZFPs$end),
                   gene_name = human_protein_coding_KZFPs$Label)


mcols(MER61E_regions)$Response <- ifelse(
  MER61E_regions$name %in% MER61E_set_3,"Acute","Non-acute")
seqlevelsStyle(KZFP_gr) <- "UCSC"
nearest_hits <- distanceToNearest(MER61E_regions,KZFP_gr, ignore.strand=TRUE)

distance_df <- data.frame(
  Peak_name = MER61E_regions$name[queryHits(nearest_hits)],
  Response = MER61E_regions$Response[queryHits(nearest_hits)],
  KZFP_gene = KZFP_gr$gene_name[subjectHits(nearest_hits)],
  Distance_bp = mcols(nearest_hits)$distance
) %>%
  mutate(
    Distance_kb = Distance_bp / 1000,
    Response = factor(Response, levels = c("Acute", "Non-acute"))
  )

ggplot(distance_df,
       aes(x = Response, y = Distance_kb, fill = Response)) +
  geom_violin(trim = FALSE, alpha = 0.5) +
  geom_boxplot(width = 0.18, outlier.shape = NA, alpha = 0.8) +
