p53_canonical_motifs <- function(fimo_root_dir, te_bed_base_dir) {
 
  
  # Initialize results list
  results_list <- list()
  hit_list <- list()
  
  # Find all directories ending with "_fimo"
  fimo_dirs <- list.dirs(fimo_root_dir, recursive = FALSE)
  fimo_dirs <- fimo_dirs[grepl("_sea_jaspar$", basename(fimo_dirs))]
  
  for (fimo_dir in fimo_dirs) {
    base_name <- sub("_sea_jaspar$", "", basename(fimo_dir))
    
    # Find FIMO file
    sites_file <- file.path(fimo_dir, "sites.tsv")
    if (!file.exists(sites_file)) {
      message("Skipping ", fimo_dir, ": sites.tsv not found")
      next
    }
  
  
  te_file <- file.path(te_bed_base_dir, paste0(base_name, ".bed"))
    if (!file.exists(te_file)) {
      message("Skipping ", base_name, ": TE BED file not found")
      next
    }
  
    
    # ------------------------------
    # Load TE BED file
    # ------------------------------
    te_df <- read.table(te_file, header = FALSE, stringsAsFactors = FALSE)
    colnames(te_df) <- c("chr","start","end","name","score","strand")
    te_df <- te_df[!is.na(te_df$start) & !is.na(te_df$end) & (te_df$start <= te_df$end), ]
    te_gr <- GRanges(
    seqnames = te_df$chr,
    ranges = IRanges(start = te_df$start, end = te_df$end),
    strand = te_df$strand,
    TE_id = te_df$name,
    TE_score = te_df$score
    )
    # ------------------------------
    # Load TP53 sites
    # ------------------------------
    cano_sites <- read.delim(sites_file, stringsAsFactors = FALSE)
    
    # Filter TP53 motifs
    if (!"motif_ALT_ID" %in% colnames(cano_sites)) {
      message("No motif_ALT_ID column in ", sites_file)
      next
    }
    cano_sites <- cano_sites[
      grepl("TP53", cano_sites$motif_ALT_ID, ignore.case = TRUE),
    ]
    # Skip if no motifs found
    if (nrow(cano_sites) == 0) {
      message("No TP53 motifs in ", base_name)
      results_list[[paste0(base_name, "_no_hits")]] <- data.frame(
        TE_name = base_name,
        motif_name = "TP53",
        total_TEs = length(te_gr),
        TEs_with_TP53 = 0,
        total_TP53_hits = 0,
        occupancy = 0
      )
      next
    }
    
    # Clean FIMO entries
    cano_sites <- cano_sites[!is.na(cano_sites$site_Start) & !is.na(cano_sites$site_End) & (cano_sites$site_Start <= cano_sites$site_End), ]
    
    motif_name <- unique(cano_sites$motif_ALT_ID)
    
    # Loop over each motif separately (in case multiple motifs exist)
    for (mn in motif_name) {
      fimo_motif <- cano_sites[cano_sites$motif_ALT_ID == mn, ]
      
      fimo_gr <- GRanges(
        seqnames = fimo_motif$seq_ID,
        ranges = IRanges(start = fimo_motif$site_Start, end = fimo_motif$site_End),
        strand = fimo_motif$site_Strand,
        score = fimo_motif$site_Score
      )
      
      # ------------------------------
      # Overlaps
      # ------------------------------
      hits <- findOverlaps(fimo_gr, te_gr)
      n_te_total <- length(te_gr)
      
      if (length(hits) == 0) {
        message("No overlaps for motif ", mn, " in ", base_name)
        results_list[[paste0(base_name, "_", mn)]] <- data.frame(
          TE_name = base_name,
          motif_name = mn,
          total_TEs = n_te_total,
          TEs_with_TP53 = 0,
          total_TP53_hits = 0,
          occupancy = 0
        )
        next
      }
      
      n_te_with_hits <- length(unique(subjectHits(hits)))
      n_total_hits <- length(hits)
      occupancy <- ifelse(n_te_total > 0, n_te_with_hits / n_te_total, NA)
      
    
    # ------------------------------
      # Compute relative positions
      # ------------------------------
      rel_pos <- (start(fimo_gr)[queryHits(hits)] - start(te_gr)[subjectHits(hits)]) /
                 width(te_gr)[subjectHits(hits)]
      rel_pos <- pmin(pmax(rel_pos, 0), 1)
      df <- data.frame(rel_pos = rel_pos)
      hit_df <- data.frame(
        TE_name = base_name,
       motif_name = mn,

    # which FIMO motif hit
       motif_chr = as.character(seqnames(fimo_gr)[queryHits(hits)]),
        motif_start = start(fimo_gr)[queryHits(hits)],
        motif_end = end(fimo_gr)[queryHits(hits)],
        motif_strand = as.character(strand(fimo_gr)[queryHits(hits)]),
        motif_score = mcols(fimo_gr)$score[queryHits(hits)],

    # which TE it overlaps
        TE_id = mcols(te_gr)$TE_id[subjectHits(hits)],
        TE_chr = as.character(seqnames(te_gr)[subjectHits(hits)]),
        TE_start = start(te_gr)[subjectHits(hits)],
        TE_end = end(te_gr)[subjectHits(hits)],
        TE_strand = as.character(strand(te_gr)[subjectHits(hits)]),

          rel_pos = rel_pos
        )

hit_list[[paste0(base_name, "_", mn)]] <- hit_df
      
      # ------------------------------
      # Store results
      # ------------------------------
      results_list[[paste0(base_name, "_", mn)]] <- data.frame(
        TE_name = base_name,
        motif_name = mn,
        total_TEs = n_te_total,
        TEs_with_TP53 = n_te_with_hits,
        total_TP53_hits = n_total_hits,
        occupancy = occupancy
      )
      
      # ------------------------------
      # Plot density
      # ------------------------------
      cat(paste0("### ", base_name, " ", mn, "\n"))
      cat(base_name, "| total TEs:", n_te_total, "| with cTP53:", n_te_with_hits, 
          "| hits:", n_total_hits, "\n")
      cat("Sequences with full TP53 motif: ", round(occupancy * 100, 2), "%\n\n")
      
      p <- ggplot(df, aes(x = rel_pos)) +
        geom_density(fill = "steelblue", alpha = 0.5) +
        labs(x = "Relative position in TE", y = "Motif density") +
        theme_classic() +
        ggtitle(paste0("canonical TP53 motif distribution: ", base_name,
                       "\nSequences with motif: ", round(occupancy * 100, 1), "%"))
      print(p)
      cat("\n\n")
    }
  } # end fimo_dirs loop
  
  # Combine all results into one data frame
  results_df <- do.call(rbind, results_list)
  
  # print(results_df)
  hits_df <- do.call(rbind, hit_list)

return(list(
  summary = results_df,
  hits = hits_df
))
}

test_pair_TE_repName <- function(df_long, rep_name, cluster1, cluster2) {
  
  # Subset for the specific repName
  sub_df <- df_long %>%
    filter(repName == rep_name) %>%
    complete(
      cluster = c(cluster1, cluster2),
      status  = c("TE", "not_TE"),
      fill = list(count = 0)
    )
  
  # fixed order of statuses
  status_levels <- c("TE", "not_TE")
  
  # make sure both statuses exist
  statuses <- unique(sub_df$status)
  if(length(statuses) != 2) {
    sub_df <- sub_df %>%
      complete(cluster, status, fill = list(count = 0))
    statuses <- unique(sub_df$status)
  }
  
  # counts for cluster1
  c1_counts <- sub_df %>%
    filter(cluster == cluster1) %>%
    arrange(factor(status, levels = status_levels)) %>%
    pull(count)
  
  # counts for cluster2
  c2_counts <- sub_df %>%
    filter(cluster == cluster2) %>%
    arrange(factor(status, levels = status_levels)) %>%
    pull(count)
  
  # 2x2 matrix for Fisher test
  mat <- matrix(
    c(c2_counts, c1_counts),
    nrow = 2,
    byrow = TRUE,
    dimnames = list(
      cluster = c(cluster2, cluster1),
      category = status_levels
    )
  )
  
  ft <- tryCatch(
    fisher.test(mat, workspace = 2e8),
    error = function(e) fisher.test(mat, simulate.p.value = TRUE, B = 1e5)
  )
  
  tibble(
    repName     = rep_name,
    comparison  = paste(cluster2, "vs", cluster1),
    odds_ratio  = ft$estimate,
    lower_CI    = ft$conf.int[1],
    upper_CI    = ft$conf.int[2],
    p_value     = ft$p.value
  )
}

prop_plot <- function(data, group_col, yes_no_col) {

  # Prepare data
  plot_data <- data %>%
    mutate(cluster = tidyr::replace_na(cluster, "NA"),
      cluster = factor(cluster,
        levels = c("Set_3", "Set_1", "not_assigned", "NA"))) %>%
    filter(cluster %in% c("Set_3", "Set_1", "not_assigned", "NA"))

  
  # Pairwise comparisons against Set_3
  comparison_groups <- c("Set_1", "not_assigned", "NA")

  test_results <- lapply(comparison_groups, function(g) {

    test_df <- plot_data %>%
      filter(cluster %in% c("Set_3", g))

    tab <- table(test_df$cluster,
              test_df[[rlang::as_name(rlang::ensym(yes_no_col))]])

    # Check expected counts
    chi <- suppressWarnings(chisq.test(tab))

    if (any(chi$expected < 5)) {
      test <- fisher.test(tab)
      method <- "Fisher"
    } else {
      test <- chi
      method <- "Chi-square"
    }

    data.frame(comparison = paste("Set_3 vs", g),
               group = g,
               p = test$p.value,
               method = method)}) %>%
    bind_rows() %>%
    mutate(p_adj = p.adjust(p, method = "BH"),
           label = paste0("p=", format.pval(p_adj, digits = 2, eps = 0.001)),
           y = c(1.10, 1.18, 1.26))


  # Proportions
  count_data <- plot_data %>%
    count({{ group_col }}, {{ yes_no_col }}) %>%
    group_by({{ group_col }}) %>%
    mutate(prop = n / sum(n)) %>%
    ungroup()


  # Plot
  ggplot(count_data,aes(x = {{ group_col }},y = prop,
      fill = {{ yes_no_col }})) +
    geom_col() +
    geom_text(aes(label = paste0(scales::percent(prop, accuracy = 1), "\n(n=", n, ")")),
              position = position_stack(vjust = 0.5)) +

    # p-values over comparison groups
    geom_text(data = test_results,aes(x = group,y = y, label = label),
              inherit.aes = FALSE,
              size = 3.5) +
    scale_y_continuous(labels = scales::percent,
                       limits = c(0, 1.3),
                       breaks = seq(0, 1, 0.25)) +
    scale_fill_manual(values =RColorBrewer::brewer.pal(12, "Paired")[1:2]) +

    labs(x = NULL,y = "Proportion") +
    theme_bw() +
    theme(axis.text.x = element_text(angle = 90,
                                     hjust = 1))
}

sox11_canonical_motifs <- function(fimo_root_dir, te_bed_base_dir) {
 
  
  # Initialize results list

    base_name <- sub("_sea_jaspar$", "", basename(fimo_dir))

    }
  
  
  te_file <- file.path(te_bed_base_dir, paste0(base_name, ".bed"))
    if (!file.exists(te_file)) {
      message("Skipping ", base_name, ": TE BED file not found")
      next
    }
  
    
    # ------------------------------
    # Load TE BED file
    # ------------------------------
    te_df <- read.table(te_file, header = FALSE, stringsAsFactors = FALSE)
    colnames(te_df) <- c("chr","start","end","name","score","strand")
    te_df <- te_df[!is.na(te_df$start) & !is.na(te_df$end) & (te_df$start <= te_df$end), ]
    te_gr <- GRanges(
    seqnames = te_df$chr,
    ranges = IRanges(start = te_df$start, end = te_df$end),
    strand = te_df$strand,
    TE_id = te_df$name,
    TE_score = te_df$score
    )
    # ------------------------------
    # Load sox11 sites
    # ------------------------------
    cano_sites <- read.delim(sites_file, stringsAsFactors = FALSE)
    
    # Filter sox11 motifs
    if (!"motif_ALT_ID" %in% colnames(cano_sites)) {
      message("No motif_ALT_ID column in ", sites_file)
      next
    }
    cano_sites <- cano_sites[
      grepl("Sox11", cano_sites$motif_ALT_ID, ignore.case = TRUE),
    ]
    # Skip if no motifs found
    if (nrow(cano_sites) == 0) {
      message("No Sox11 motifs in ", base_name)
      results_list[[paste0(base_name, "_no_hits")]] <- data.frame(
        TE_name = base_name,
        motif_name = "Sox11",
        total_TEs = length(te_gr),
        TEs_with_Sox11 = 0,
        total_Sox11_hits = 0,
        occupancy = 0
      )
      next
    }
    
    # Clean FIMO entries
    cano_sites <- cano_sites[!is.na(cano_sites$site_Start) & !is.na(cano_sites$site_End) & (cano_sites$site_Start <= cano_sites$site_End), ]
    
    motif_name <- unique(cano_sites$motif_ALT_ID)
    
    # Loop over each motif separately (in case multiple motifs exist)
    for (mn in motif_name) {
      fimo_motif <- cano_sites[cano_sites$motif_ALT_ID == mn, ]
      
      fimo_gr <- GRanges(
        seqnames = fimo_motif$seq_ID,
        ranges = IRanges(start = fimo_motif$site_Start, end = fimo_motif$site_End),
        strand = fimo_motif$site_Strand,
        score = fimo_motif$site_Score
      )
      
      # ------------------------------
      # Overlaps
      # ------------------------------
      hits <- findOverlaps(fimo_gr, te_gr)
      n_te_total <- length(te_gr)
      
      if (length(hits) == 0) {
        message("No overlaps for motif ", mn, " in ", base_name)
        results_list[[paste0(base_name, "_", mn)]] <- data.frame(
          TE_name = base_name,
          motif_name = mn,
          total_TEs = n_te_total,
          TEs_with_Sox11 = 0,
          total_Sox11_hits = 0,
          occupancy = 0
        )
        next
      }
      
      n_te_with_hits <- length(unique(subjectHits(hits)))
      n_total_hits <- length(hits)
      occupancy <- ifelse(n_te_total > 0, n_te_with_hits / n_te_total, NA)
      
    
    # ------------------------------
      # Compute relative positions
      # ------------------------------
      rel_pos <- (start(fimo_gr)[queryHits(hits)] - start(te_gr)[subjectHits(hits)]) /
                 width(te_gr)[subjectHits(hits)]
      rel_pos <- pmin(pmax(rel_pos, 0), 1)
      df <- data.frame(rel_pos = rel_pos)
      hit_df <- data.frame(
        TE_name = base_name,
       motif_name = mn,

    # which FIMO motif hit
       motif_chr = as.character(seqnames(fimo_gr)[queryHits(hits)]),
        motif_start = start(fimo_gr)[queryHits(hits)],
        motif_end = end(fimo_gr)[queryHits(hits)],
        motif_strand = as.character(strand(fimo_gr)[queryHits(hits)]),
        motif_score = mcols(fimo_gr)$score[queryHits(hits)],

    # which TE it overlaps
        TE_id = mcols(te_gr)$TE_id[subjectHits(hits)],
        TE_chr = as.character(seqnames(te_gr)[subjectHits(hits)]),
        TE_start = start(te_gr)[subjectHits(hits)],
        TE_end = end(te_gr)[subjectHits(hits)],
        TE_strand = as.character(strand(te_gr)[subjectHits(hits)]),

          rel_pos = rel_pos
        )

hit_list[[paste0(base_name, "_", mn)]] <- hit_df
      
      # ------------------------------
      # Store results
      # ------------------------------
      results_list[[paste0(base_name, "_", mn)]] <- data.frame(
        TE_name = base_name,
        motif_name = mn,
        total_TEs = n_te_total,
        TEs_with_Sox11 = n_te_with_hits,
        total_Sox11_hits = n_total_hits,
        occupancy = occupancy
      )
      
      # ------------------------------
      # Plot density
      # ------------------------------
      cat(paste0("### ", base_name, " ", mn, "\n"))
      cat(base_name, "| total TEs:", n_te_total, "| with Sox11:", n_te_with_hits, 
          "| hits:", n_total_hits, "\n")
      cat("Sequences with full Sox11 motif: ", round(occupancy * 100, 2), "%\n\n")
      
      p <- ggplot(df, aes(x = rel_pos)) +
        geom_density(fill = "steelblue", alpha = 0.5) +
        labs(x = "Relative position in TE", y = "Motif density") +
        theme_classic() +
        ggtitle(paste0("canonical Sox11 motif distribution: ", base_name,
                       "\nSequences with motif: ", round(occupancy * 100, 1), "%"))
      print(p)
      cat("\n\n")
    }
  } # end fimo_dirs loop
  
  # Combine all results into one data frame
  results_df <- do.call(rbind, results_list)
  
  # print(results_df)
  hits_df <- do.call(rbind, hit_list)

return(list(
  summary = results_df,
  hits = hits_df
))
}
