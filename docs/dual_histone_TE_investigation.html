run_family_fisher <- function(df,
                              a = "n_H3K27ac",
                              b = "not_H3K27ac",
                              c = "n_H3K9me3",
                              d = "not_H3K9me3",
                              family_col = "repFamily") {

  res <- lapply(seq_len(nrow(df)), function(i) {

    x <- df[i, ]

    tab <- matrix(
      c(x[[a]], x[[b]],
        x[[c]], x[[d]]),
      nrow = 2,
      byrow = TRUE
    )

    ft <- fisher.test(tab)

    data.frame(
      repFamily = x[[family_col]],
      odds_ratio = as.numeric(ft$estimate),
      conf_low = ft$conf.int[1],
      conf_high = ft$conf.int[2],
      p_value = ft$p.value
    )
  })

  out <- do.call(rbind, res)

  out$FDR <- p.adjust(out$p_value, method = "BH")

  out
}
run_name_fisher <- function(df,
                              a = "n_H3K27ac",
                              b = "not_H3K27ac",
                              c = "n_H3K9me3",
                              d = "not_H3K9me3",
                              family_col = "repName") {

  res <- lapply(seq_len(nrow(df)), function(i) {

    x <- df[i, ]

    tab <- matrix(
      c(x[[a]], x[[b]],
        x[[c]], x[[d]]),
      nrow = 2,
      byrow = TRUE
    )

    ft <- fisher.test(tab)

    data.frame(
      repName = x[[family_col]],
      odds_ratio = as.numeric(ft$estimate),
      conf_low = ft$conf.int[1],
      conf_high = ft$conf.int[2],
      p_value = ft$p.value
    )
  })

  out <- do.call(rbind, res)

  out$FDR <- p.adjust(out$p_value, method = "BH")

  out
}
