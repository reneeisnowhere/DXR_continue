p53_canonical_motifs <- function(fimo_root_dir, te_bed_base_dir) {
 
  
  # Initialize results list
  results_list <- list()
  
  # Find all directories ending with "_fimo"
  fimo_dirs <- list.dirs(fimo_root_dir, recursive = FALSE)
  fimo_dirs <- fimo_dirs[grepl("_sea_jaspar$", basename(fimo_dirs))]
  
  for (fimo_dir in fimo_dirs) {
    base_name <- sub("_sea_jaspar$", "", basename(fimo_dir))
    
    # Find FIMO file
    sites_file <- file.path(fimo_dir, "sites.tsv")
    if (!file.exists(sites_file)) {
      message("Skipping ", fimo_dir, ": sites.tsv not found")
      next
    }
  
  
  te_file <- file.path(te_bed_base_dir, paste0(base_name, ".bed"))
    if (!file.exists(te_file)) {
      message("Skipping ", base_name, ": TE BED file not found")
      next
    }
  
    
    # ------------------------------
    # Load TE BED file
    # ------------------------------
    te_df <- read.table(te_file, header = FALSE, stringsAsFactors = FALSE)
    colnames(te_df) <- c("chr","start","end","name","score","strand")
    te_df <- te_df[!is.na(te_df$start) & !is.na(te_df$end) & (te_df$start <= te_df$end), ]
    te_gr <- GRanges(
      seqnames = te_df$chr,
      ranges = IRanges(start = te_df$start, end = te_df$end),
      strand = te_df$strand
    )
    # ------------------------------
    # Load TP53 sites
    # ------------------------------
    cano_sites <- read.delim(sites_file, stringsAsFactors = FALSE)
    
    # Filter TP53 motifs
    if (!"motif_ALT_ID" %in% colnames(cano_sites)) {
      message("No motif_ALT_ID column in ", sites_file)
      next
    }
    cano_sites <- cano_sites[
      grepl("TP53", cano_sites$motif_ALT_ID, ignore.case = TRUE),
    ]
    # Skip if no motifs found
    if (nrow(cano_sites) == 0) {
      message("No TP53 motifs in ", base_name)
      results_list[[paste0(base_name, "_no_hits")]] <- data.frame(
        TE_name = base_name,
        motif_name = "TP53",
        total_TEs = length(te_gr),
        TEs_with_TP53 = 0,
        total_TP53_hits = 0,
        occupancy = 0
      )
      next
    }
    
    # Clean FIMO entries
    cano_sites <- cano_sites[!is.na(cano_sites$site_Start) & !is.na(cano_sites$site_End) & (cano_sites$site_Start <= cano_sites$site_End), ]
    
    motif_name <- unique(cano_sites$motif_ALT_ID)
    
    # Loop over each motif separately (in case multiple motifs exist)
    for (mn in motif_name) {
      fimo_motif <- cano_sites[cano_sites$motif_ALT_ID == mn, ]
      
      fimo_gr <- GRanges(
        seqnames = fimo_motif$seq_ID,
        ranges = IRanges(start = fimo_motif$site_Start, end = fimo_motif$site_End),
        strand = fimo_motif$site_Strand,
        score = fimo_motif$site_Score
      )
      
      # ------------------------------
      # Overlaps
      # ------------------------------
      hits <- findOverlaps(fimo_gr, te_gr)
      n_te_total <- length(te_gr)
      
      if (length(hits) == 0) {
        message("No overlaps for motif ", mn, " in ", base_name)
        results_list[[paste0(base_name, "_", mn)]] <- data.frame(
          TE_name = base_name,
          motif_name = mn,
          total_TEs = n_te_total,
          TEs_with_TP53 = 0,
          total_TP53_hits = 0,
          occupancy = 0
        )
        next
      }
      
      n_te_with_hits <- length(unique(subjectHits(hits)))
      n_total_hits <- length(hits)
      occupancy <- ifelse(n_te_total > 0, n_te_with_hits / n_te_total, NA)
      
    
    # ------------------------------
      # Compute relative positions
      # ------------------------------
      rel_pos <- (start(fimo_gr)[queryHits(hits)] - start(te_gr)[subjectHits(hits)]) /
                 width(te_gr)[subjectHits(hits)]
      rel_pos <- pmin(pmax(rel_pos, 0), 1)
      df <- data.frame(rel_pos = rel_pos)
      
      # ------------------------------
      # Store results
      # ------------------------------
      results_list[[paste0(base_name, "_", mn)]] <- data.frame(
        TE_name = base_name,
        motif_name = mn,
        total_TEs = n_te_total,
        TEs_with_TP53 = n_te_with_hits,
        total_TP53_hits = n_total_hits,
        occupancy = occupancy
      )
      
      # ------------------------------
      # Plot density
      # ------------------------------
      cat(paste0("### ", base_name, " ", mn, "\n"))
      cat(base_name, "| total TEs:", n_te_total, "| with cTP53:", n_te_with_hits, 
          "| hits:", n_total_hits, "\n")
      cat("Sequences with full TP53 motif: ", round(occupancy * 100, 2), "%\n\n")
      
      p <- ggplot(df, aes(x = rel_pos)) +
        geom_density(fill = "steelblue", alpha = 0.5) +
        labs(x = "Relative position in TE", y = "Motif density") +
        theme_classic() +
        ggtitle(paste0("canonical TP53 motif distribution: ", base_name,
                       "\nSequences with motif: ", round(occupancy * 100, 1), "%"))
      print(p)
      cat("\n\n")
    }
  } # end fimo_dirs loop
  
  # Combine all results into one data frame
  results_df <- do.call(rbind, results_list)
  
  print(results_df)
  return(results_df)
}

plot_p53_motifs_all <- function(fimo_root_dir, te_bed_base_dir) {
 
  
  # Initialize results list
  results_list <- list()
  detail_list <- list()
  # Find all directories ending with "_fimo"
  fimo_dirs <- list.dirs(fimo_root_dir, recursive = FALSE)
  fimo_dirs <- fimo_dirs[grepl("_fimo$", basename(fimo_dirs))]

    if (!file.exists(fimo_file)) {
      message("Skipping ", fimo_dir, ": fimo.tsv not found")
      next
    }
    
    # Find corresponding TE BED file
    te_file <- file.path(te_bed_base_dir, paste0(base_name, ".bed"))
    if (!file.exists(te_file)) {
      message("Skipping ", base_name, ": TE BED file not found")
      next
    }
    
    # ------------------------------
    # Load TE BED file
    # ------------------------------
    te_df <- read.table(te_file, header = FALSE, stringsAsFactors = FALSE)
    colnames(te_df) <- c("chr","start","end","name","score","strand")
    te_df <- te_df[!is.na(te_df$start) & !is.na(te_df$end) & (te_df$start <= te_df$end), ]
    te_gr <- GRanges(
      seqnames = te_df$chr,
      ranges = IRanges(start = te_df$start, end = te_df$end),
      strand = te_df$strand,
      name = te_df$name)
    
    
    # ------------------------------
    # Load FIMO
    # ------------------------------
    fimo <- read.delim(fimo_file, stringsAsFactors = FALSE)
    
    # Filter TP53 motifs
    if ("motif_id" %in% colnames(fimo)) {
      fimo <- fimo[grepl("TP53", fimo$motif_id, ignore.case = TRUE), ]
    } else {
      message("No motif_id column in ", fimo_file)
      next
    }
    
    # Skip if no motifs found
    if (nrow(fimo) == 0) {
      message("No TP53 motifs in ", base_name)
      results_list[[paste0(base_name, "_no_hits")]] <- data.frame(
        TE_name = base_name,
        motif_name = NA,
        total_TEs = length(te_gr),
        TEs_with_TP53 = 0,
        total_TP53_hits = 0,
        occupancy = 0
      )
      next
    }
    
    # Clean FIMO entries
    fimo <- fimo[!is.na(fimo$start) & !is.na(fimo$stop) & (fimo$start <= fimo$stop), ]
    
    motif_name <- unique(fimo$motif_id)
    
    # Loop over each motif separately (in case multiple motifs exist)
    for (mn in motif_name) {
      fimo_motif <- fimo[fimo$motif_id == mn, ]
      
      fimo_gr <- GRanges(
        seqnames = fimo_motif$sequence_name,
        ranges = IRanges(start = fimo_motif$start, end = fimo_motif$stop),
        strand = fimo_motif$strand,
        score = fimo_motif$score
      )
      
      # ------------------------------
      # Overlaps
      # ------------------------------
      hits <- findOverlaps(fimo_gr, te_gr)
      n_te_total <- length(te_gr)
      
      if (length(hits) == 0) {
        message("No overlaps for motif ", mn, " in ", base_name)
        results_list[[paste0(base_name, "_", mn)]] <- data.frame(
          TE_name = base_name,
          motif_name = mn,
          total_TEs = n_te_total,
          TEs_with_TP53 = 0,
          total_TP53_hits = 0,
          occupancy = 0
        )
        next
      }
      
      n_te_with_hits <- length(unique(subjectHits(hits)))
      n_total_hits <- length(hits)
      occupancy <- n_te_with_hits / n_te_total
      te_names <- mcols(te_gr)$name[subjectHits(hits)]
     
      
      # ------------------------------
      # Compute relative positions
      # ------------------------------
      rel_pos <- (start(fimo_gr)[queryHits(hits)] - start(te_gr)[subjectHits(hits)]) /
                 width(te_gr)[subjectHits(hits)]
      rel_pos <- pmin(pmax(rel_pos, 0), 1)
      df <- data.frame(rel_pos = rel_pos)
      
      # ------------------------------
      # Store results
      # ------------------------------
      results_list[[paste0(base_name, "_", mn)]] <- data.frame(
        TE_name = base_name,
        motif_name = mn,
        total_TEs = n_te_total,
        TEs_with_TP53 = n_te_with_hits,
        total_TP53_hits = n_total_hits,
        occupancy = occupancy
      )
      # ------------------------------
# Store details ONLY if hits exist
# ------------------------------
if (length(hits) > 0 && length(subjectHits(hits)) > 0) {

  detail_list[[paste0(base_name, "_", mn)]] <- data.frame(
    TE_name = base_name,
    motif_name = mn,
    Peakid = mcols(te_gr)$name[subjectHits(hits)]
  )

}
      
      # ------------------------------
      # Plot density
      # ------------------------------
      cat(paste0("### ", base_name, " ", mn, "\n"))
      cat(base_name, "| total TEs:", n_te_total, "| with TP53:", n_te_with_hits, 
          "| hits:", n_total_hits, "\n")
      cat("Sequences with TP53 motif: ", round(occupancy * 100, 2), "%\n\n")
      
      p <- ggplot(df, aes(x = rel_pos)) +
        geom_density(fill = "steelblue", alpha = 0.5) +
        labs(x = "Relative position in TE", y = "Motif density") +
        theme_classic() +
        ggtitle(paste0("TP53 motif distribution: ", base_name,
                       "\nSequences with motif: ", round(occupancy * 100, 1), "%"))
      print(p)
      cat("\n\n")
    }
  } # end fimo_dirs loop
  
  # Combine all results into one data frame
  results_df <- do.call(rbind, results_list)
  detail_df <- dplyr::bind_rows(detail_list)
  print(results_df)
  return(list(
  summary = results_df,
  detail  = detail_df
))
}

no_p53_results <- lapply(Mer_list, function(m) {

  df_sub <- df_S1n3 %>%
    filter(repName == m) %>%
    mutate(
      cluster = factor(cluster, levels = c("Set_1","Set_3")),
      no_p53 = factor(p53_bin == 0, levels = c(FALSE, TRUE))
    )
  
  tab <- table(df_sub$cluster, df_sub$no_p53)
  
   # skip if not full 2x2
  if (any(dim(tab) < 2) || any(rowSums(tab) == 0) || any(colSums(tab) == 0)) {
    return(data.frame(
      repName = m,
      odds_ratio = NA,
      conf_low = NA,
      conf_high = NA,
      p_value = NA,
      note = "insufficient data"
    ))
  }

  ft <- fisher.test(tab)

  data.frame(
    repName = m,
    odds_ratio = unname(ft$estimate),
    conf_low = ft$conf.int[1],
    conf_high = ft$conf.int[2],
   p_value = ft$p.value,
    note = "ok"
  )
})
