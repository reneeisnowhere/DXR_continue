plot_fimo_motifs <- function(fimo_root_dir, te_bed_file) {
  
  # # Create output dir if it doesn't exist
  # dir.create(out_dir, showWarnings = FALSE, recursive = TRUE)
  # 
  # ------------------------------
  # Load TE BED file
  # ------------------------------
  te_df <- read.table(te_bed_file, header = FALSE, stringsAsFactors = FALSE)
  colnames(te_df) <- c("chr","start","end","name","score","strand")
  te_df <- te_df[!is.na(te_df$start) & !is.na(te_df$end) & (te_df$start <= te_df$end), ]
  te_gr <- GRanges(
    seqnames = te_df$chr,
    ranges = IRanges(start = te_df$start, end = te_df$end),
    strand = te_df$strand
  )
  
  # ------------------------------
  # Find all FIMO directories
  # ------------------------------
  fimo_dirs <- list.dirs(fimo_root_dir, recursive = FALSE)
  
  for (fimo_subdir in fimo_dirs) {
    
    fimo_file <- file.path(fimo_subdir, "fimo.tsv")
    
    if (!file.exists(fimo_file)) {
      message("Skipping ", fimo_subdir, ": fimo.tsv not found")
      next
    }
    
    message("Processing ", fimo_file)
    
    # ------------------------------
    # Load and clean FIMO
    # ------------------------------
    fimo <- read.delim(fimo_file, stringsAsFactors = FALSE)
    fimo <- fimo[!is.na(fimo$start) & !is.na(fimo$stop) & (fimo$start <= fimo$stop), ]
    motif_name <- unique(fimo$motif_alt_id)
    
    if (length(motif_name) > 1) {motif_name <- motif_name[1]}

    
    if (nrow(fimo) == 0) {
      message("No valid FIMO entries for ", fimo_subdir)
      next
    }
    
    fimo_gr <- GRanges(
      seqnames = fimo$sequence_name,
      ranges = IRanges(start = fimo$start, end = fimo$stop),
      strand = fimo$strand,
      score = fimo$score
    )
    
    # ------------------------------
    # Find overlaps with TEs
    # ------------------------------
    hits <- findOverlaps(fimo_gr, te_gr)
    
    if (length(hits) == 0) {
      message("No overlaps found for ", fimo_subdir)
      next
    }
    
    # ------------------------------
    # Compute relative positions
    # ------------------------------
    rel_pos <- (start(fimo_gr)[queryHits(hits)] - start(te_gr)[subjectHits(hits)]) /
               width(te_gr)[subjectHits(hits)]
    rel_pos <- pmin(pmax(rel_pos, 0), 1)
    
    df <- data.frame(rel_pos = rel_pos)
    
    # ------------------------------
    # Plot density
    # ------------------------------
   
    cat(paste0("### MER61E ", motif_name, "\n\n"))
    p <- ggplot(df, aes(x = rel_pos)) +
      geom_density(fill = "steelblue", alpha = 0.5) +
      labs(x = "Relative position in TE", y = "Motif density") +
      theme_classic() +
      ggtitle(paste("Motif distribution:", basename(fimo_subdir),"\n",motif_name))
    
   print(p)
   cat("\n\n")
  }
  
 
}

plot_p53_motifs_all <- function(fimo_root_dir, te_bed_base_dir) {
 
  
  # Initialize results list
  results_list <- list()
  
  # Find all directories ending with "_fimo"
  fimo_dirs <- list.dirs(fimo_root_dir, recursive = FALSE)
  fimo_dirs <- fimo_dirs[grepl("_fimo$", basename(fimo_dirs))]
  
  for (fimo_dir in fimo_dirs) {
    base_name <- sub("_fimo$", "", basename(fimo_dir))
    
    # Find FIMO file
    fimo_file <- file.path(fimo_dir, "fimo.tsv")
    if (!file.exists(fimo_file)) {
      message("Skipping ", fimo_dir, ": fimo.tsv not found")
      next
    }
    
    # Find corresponding TE BED file
    te_file <- file.path(te_bed_base_dir, paste0(base_name, ".bed"))
    if (!file.exists(te_file)) {
      message("Skipping ", base_name, ": TE BED file not found")
      next
    }
    
    # ------------------------------
    # Load TE BED file
    # ------------------------------
    te_df <- read.table(te_file, header = FALSE, stringsAsFactors = FALSE)
    colnames(te_df) <- c("chr","start","end","name","score","strand")
    te_df <- te_df[!is.na(te_df$start) & !is.na(te_df$end) & (te_df$start <= te_df$end), ]
    te_gr <- GRanges(
      seqnames = te_df$chr,
      ranges = IRanges(start = te_df$start, end = te_df$end),
      strand = te_df$strand
    )
    
    # ------------------------------
    # Load FIMO
    # ------------------------------
    fimo <- read.delim(fimo_file, stringsAsFactors = FALSE)
    
    # Filter TP53 motifs
    if ("motif_id" %in% colnames(fimo)) {
      fimo <- fimo[grepl("TP53", fimo$motif_id, ignore.case = TRUE), ]
    } else {
      message("No motif_id column in ", fimo_file)
      next
    }
    
    # Skip if no motifs found
    if (nrow(fimo) == 0) {
      message("No TP53 motifs in ", base_name)
      results_list[[paste0(base_name, "_no_hits")]] <- data.frame(
        TE_name = base_name,
        motif_name = NA,
        total_TEs = length(te_gr),
        TEs_with_TP53 = 0,
        total_TP53_hits = 0,
        occupancy = 0
      )
      next
    }
    
    # Clean FIMO entries
    fimo <- fimo[!is.na(fimo$start) & !is.na(fimo$stop) & (fimo$start <= fimo$stop), ]
    
    motif_name <- unique(fimo$motif_id)
    
    # Loop over each motif separately (in case multiple motifs exist)
    for (mn in motif_name) {
      fimo_motif <- fimo[fimo$motif_id == mn, ]
      
      fimo_gr <- GRanges(
        seqnames = fimo_motif$sequence_name,
        ranges = IRanges(start = fimo_motif$start, end = fimo_motif$stop),
        strand = fimo_motif$strand,
        score = fimo_motif$score
      )
      
      # ------------------------------
      # Overlaps
      # ------------------------------
      hits <- findOverlaps(fimo_gr, te_gr)
      n_te_total <- length(te_gr)
      
      if (length(hits) == 0) {
        message("No overlaps for motif ", mn, " in ", base_name)
        results_list[[paste0(base_name, "_", mn)]] <- data.frame(
          TE_name = base_name,
          motif_name = mn,
          total_TEs = n_te_total,
          TEs_with_TP53 = 0,
          total_TP53_hits = 0,
          occupancy = 0
        )
        next
      }
      
      n_te_with_hits <- length(unique(subjectHits(hits)))
      n_total_hits <- length(hits)
      occupancy <- n_te_with_hits / n_te_total
      
      # ------------------------------
      # Compute relative positions
      # ------------------------------
      rel_pos <- (start(fimo_gr)[queryHits(hits)] - start(te_gr)[subjectHits(hits)]) /
                 width(te_gr)[subjectHits(hits)]
      rel_pos <- pmin(pmax(rel_pos, 0), 1)
      df <- data.frame(rel_pos = rel_pos)
      
      # ------------------------------
      # Store results
      # ------------------------------
      results_list[[paste0(base_name, "_", mn)]] <- data.frame(
        TE_name = base_name,
        motif_name = mn,
        total_TEs = n_te_total,
        TEs_with_TP53 = n_te_with_hits,
        total_TP53_hits = n_total_hits,
        occupancy = occupancy
      )
      
      # ------------------------------
      # Plot density
      # ------------------------------
      cat(paste0("### ", base_name, " ", mn, "\n"))
      cat(base_name, "| total TEs:", n_te_total, "| with TP53:", n_te_with_hits, 
          "| hits:", n_total_hits, "\n")
      cat("Sequences with TP53 motif: ", round(occupancy * 100, 2), "%\n\n")
      
      p <- ggplot(df, aes(x = rel_pos)) +
        geom_density(fill = "steelblue", alpha = 0.5) +
        labs(x = "Relative position in TE", y = "Motif density") +
        theme_classic() +
        ggtitle(paste0("TP53 motif distribution: ", base_name,
                       "\nSequences with motif: ", round(occupancy * 100, 1), "%"))
      print(p)
      cat("\n\n")
    }
  } # end fimo_dirs loop
  
  # Combine all results into one data frame
  results_df <- do.call(rbind, results_list)
  
  print(results_df)
  return(results_df)
}
