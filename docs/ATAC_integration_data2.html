  as.data.frame() %>%

  left_join(H3K9me3_lookup) %>% 
  mutate(cluster = if_else(is.na(cluster), "NA", cluster)) %>% 
  mutate(
    ATAC_type = case_when(Peakid %in% joint_H3K9me3 ~ "joint",
      Peakid %in% H3K9me3_CAR_peakid$ROI_ID ~ "CAR",
      Peakid %in% H3K9me3_DAR_peakid$ROI_ID ~ "DAR",
      TRUE ~ "not_atac"
    )
  ) %>%
  count(cluster, ATAC_type, name = "n") %>%
  group_by(cluster) %>%
  mutate(
    percent = n / sum(n) * 100
  ) %>%
  ungroup()
